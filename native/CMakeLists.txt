cmake_minimum_required(VERSION 3.22)
project(ecologist LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_library(ecologist_core STATIC core/world.cpp core/grains.cpp core/chemistry.cpp core/sandbox.cpp core/machines.cpp core/material_systems.cpp core/reaction_systems.cpp)
if(MSVC)
  target_compile_options(ecologist_core PRIVATE /fp:strict)
else()
  target_compile_options(ecologist_core PRIVATE -ffp-contract=off -fno-fast-math)
endif()
add_executable(ecologist-core core/cli.cpp)
add_library(ecologist_render STATIC render/cells.cpp)
target_link_libraries(ecologist_render PUBLIC ecologist_core)
target_link_libraries(ecologist-core PRIVATE ecologist_render)
add_executable(ecologist-chemistry-tests core/chemistry_tests.cpp)
target_link_libraries(ecologist-chemistry-tests PRIVATE ecologist_core)
add_executable(ecologist-sandbox-tests core/sandbox_tests.cpp)
target_link_libraries(ecologist-sandbox-tests PRIVATE ecologist_core)
option(ECOLOGIST_GODOT "Build the Godot extension" ON)
if(ECOLOGIST_GODOT)
  if(NOT GODOT_CPP_DIR)
    message(FATAL_ERROR "Set GODOT_CPP_DIR to the pinned godot-cpp checkout; see tools/native.ps1")
  endif()
  set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build-profile.json" CACHE FILEPATH "" FORCE)
  add_subdirectory("${GODOT_CPP_DIR}" godot-cpp EXCLUDE_FROM_ALL)
  if(MINGW)
    # godot-cpp 4.5 relies on a transitive C allocation header missing in newer libc++.
    target_compile_options(godot-cpp PRIVATE -include cstdlib)
  endif()
  add_library(ecologist SHARED godot/ecologist_world.cpp)
  target_link_libraries(ecologist PRIVATE ecologist_render godot-cpp)
  set_target_properties(ecologist PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../godot/bin" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../godot/bin")
  if(MINGW)
    target_link_options(ecologist PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
if(MINGW)
  target_link_options(ecologist-core PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(ecologist-chemistry-tests PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(ecologist-sandbox-tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
